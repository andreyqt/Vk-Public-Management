#Example of procedure(execution) on vk script language

var domain = Args.domain;
var query = Args.query;
var begin = Args.begin;
var end = Args.end;
var token = "my_token";

var count = 0;
var offset = 0;
var requests = 1;
var remaining = 25;

while (remaining != 0) {
var response = API.wall.search({
"access_token": token,
"query": query,
"domain": domain,
"count": 100,
"offset": offset
}).items@.date;

int i = 0;
while (i < response.length) {
if (response[i] >= begin && response[i] <= end) {
count = count + 1;
}
i = i + 1;
}

if (response.length < 100 || response[response.length-1] < begin) {
return {
"count": count,
"offset": offset,
"requests": requests,
"message": "success"
};
}

offset = offset + 100;
remaining = remaining - 1;
}

return {
"count": count,
"requests": requests,
"offset": offset,
"message": "api limit reached"
};
